cmake_minimum_required(VERSION 3.16)
project(piensaware_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LV_CONF_SKIP ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../components/lvgl" lvgl)
target_compile_definitions(lvgl PUBLIC LV_CONF_SKIP=1 LV_USE_OS=0 LV_MEM_SIZE=16777216 LV_LVGL_H_INCLUDE_SIMPLE=1 LV_FONT_MONTSERRAT_18=1 LV_FONT_MONTSERRAT_20=1 LV_FONT_MONTSERRAT_24=1 LV_FONT_MONTSERRAT_28=1 LV_FONT_MONTSERRAT_32=1 LV_FONT_MONTSERRAT_40=1 LV_FONT_MONTSERRAT_48=1)
file(GLOB FONTS "../main/fonts/poppins_*.c")
add_executable(preview preview.cpp ../main/ui.cpp ../main/assets/logo.c ${FONTS})
target_compile_definitions(preview PRIVATE PIENSAWARE_HOST=1)
target_compile_options(preview PRIVATE -UNDEBUG)
target_include_directories(preview PRIVATE . stubs ../main)
target_link_libraries(preview PRIVATE lvgl)
